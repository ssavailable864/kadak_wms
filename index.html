// TABS SWITCH LOGIC function switchTab(panelId) { document.querySelectorAll('.page-panel').forEach(p => p.classList.remove('active')); document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active')); document.getElementById(panelId).classList.add('active'); if(event) event.target.classList.add('active'); if(panelId === 'orders-panel') loadOrdersData(); } // FETCH ORDERS async function loadOrdersData() { let response = await fetch('http://127.0.0.1:8000/api/orders'); let orders = await response.json(); let tbody = document.getElementById('orders-rows'); tbody.innerHTML = ''; orders.forEach(o => { let badgeClass = o.channel === 'Amazon' ? 'badge-amazon' : 'badge-meesho'; tbody.innerHTML += ` ${o.order_no} ${o.channel} ${o.sku} ${o.status} `; }); } // TRIGGER PICKLIST async function triggerPicklistGeneration() { let response = await fetch('http://127.0.0.1:8000/api/picklist-preview'); let data = await response.json(); switchTab('picklist-panel'); let tbody = document.getElementById('picklist-rows'); tbody.innerHTML = ''; data.items.forEach(i => { tbody.innerHTML += ` ${i.order_no} ${i.sku} ${i.qty} ${i.rack} ${i.type} `; }); } // ASALI EXCEL UPLOAD FRONTEND FUNCTION async function uploadExcelFile() { let fileInput = document.getElementById('excelFile'); if(fileInput.files.length === 0) { alert("Bhai pahle koi Excel file select toh karo!"); return; } let file = fileInput.files[0]; let formData = new FormData(); formData.append("file", file); try { let response = await fetch('http://127.0.0.1:8000/api/upload-mapping', { method: 'POST', body: formData }); let result = await response.json(); if(response.ok) { alert(result.message); // Success Message dikhayega pop-up me switchTab('orders-panel'); // Automatic order panel par le jayega } else { alert("Error: " + result.detail); } } catch (error) { alert("Bhai, backend server se connect nahi ho pa rha hai!"); } } // INITIAL LOAD loadOrdersData();
